  <script>
    // --- TYPEWRITER EFFECT FOR TERMINAL ---
    const codeLines = [
      { text: "// Initializing profile...", class: "comment" },
      { text: "package main", class: "kw" },
      { text: "import", class: "kw" },
      { text: '  "fmt"', class: "str" },
      { text: "", class: "" },
      { text: "func main() {", class: "fn" },
      { text: '  fmt.Println("Hello World!")', class: "" },
      { text: "}", class: "" }
    ];

    const typeContainer = document.getElementById('typewriter');
    let lineIndex = 0;
    let charIndex = 0;

    function typeLine() {
      if (lineIndex < codeLines.length) {
        // Create new line div if starting a new line
        let currentLineDiv = document.querySelector(`.line-${lineIndex}`);
        if (!currentLineDiv) {
            currentLineDiv = document.createElement('div');
            currentLineDiv.classList.add(`line-${lineIndex}`);
            currentLineDiv.style.minHeight = "20px";
            if(codeLines[lineIndex].class) currentLineDiv.classList.add(codeLines[lineIndex].class);
            typeContainer.appendChild(currentLineDiv);
        }

        const currentText = codeLines[lineIndex].text;

        if (charIndex < currentText.length) {
          currentLineDiv.textContent += currentText.charAt(charIndex);
          charIndex++;
          setTimeout(typeLine, Math.random() * 30 + 30); // Random typing speed
        } else {
          lineIndex++;
          charIndex = 0;
          setTimeout(typeLine, 150); // Pause between lines
        }
      }
    }

    // Start typing after a small delay
    setTimeout(typeLine, 1000);

    // --- SCROLL REVEAL ANIMATION ---
    const revealElements = document.querySelectorAll('.reveal');

    const revealOnScroll = () => {
      const windowHeight = window.innerHeight;
      const elementVisible = 150;

      revealElements.forEach((reveal) => {
        const elementTop = reveal.getBoundingClientRect().top;
        if (elementTop < windowHeight - elementVisible) {
          reveal.classList.add('active');
        }
      });
    };

    window.addEventListener('scroll', revealOnScroll);
    revealOnScroll(); // Trigger once on load
  </script>
